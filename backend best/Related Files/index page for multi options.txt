    const API_BASE = window.location.origin;      // serves Flask backend on same host/port

    // Mapping of signal types to CSS class & display text
    function getSignalClass(signalType) {
        if (!signalType) return 'signal-NO_TRADE';
        if (signalType === 'STRONG_BUY_CE') return 'signal-STRONG_BUY_CE';
        if (signalType === 'BUY_CE') return 'signal-BUY_CE';
        if (signalType === 'LOW_BUY_CE') return 'signal-LOW_BUY_CE';
        if (signalType === 'STRONG_BUY_PE') return 'signal-STRONG_BUY_PE';
        if (signalType === 'BUY_PE') return 'signal-BUY_PE';
        if (signalType === 'LOW_BUY_PE') return 'signal-LOW_BUY_PE';
        if (signalType === 'ACTIVE') return 'signal-ACTIVE';
        return 'signal-NO_TRADE';
    }

    function formatSignalText(signalType) {
        if (!signalType) return 'NO TRADE';
        if (signalType === 'STRONG_BUY_CE') return '🔥 STRONG BUY CE';
        if (signalType === 'BUY_CE') return '✔️ BUY CE';
        if (signalType === 'LOW_BUY_CE') return '🟡 LOW BUY CE';
        if (signalType === 'STRONG_BUY_PE') return '🔥 STRONG BUY PE';
        if (signalType === 'BUY_PE') return '✔️ BUY PE';
        if (signalType === 'LOW_BUY_PE') return '🟠 LOW BUY PE';
        if (signalType === 'ACTIVE') return '⏳ ACTIVE TRADE';
        return '⚪ NO TRADE';
    }

    function getTrendClass(trend) {
        if (trend === 'BULLISH') return 'bullish';
        if (trend === 'BEARISH') return 'bearish';
        return '';
    }

    function formatNumber(num, decimals = 2) {
        if (num === undefined || num === null) return '—';
        return Number(num).toFixed(decimals);
    }

    // Render all index cards
    function renderDashboard(data) {
        const signals = data.signals || {};
        const sentiment = data.sentiment || {};
        const portfolios = data.portfolios || {};
        const container = document.getElementById('dashboardContainer');
        if (!container) return;

        // Order of indices
        const indices = ['NIFTY', 'BANKNIFTY', 'FINNIFTY', 'MIDCPNIFTY', 'SENSEX'];
        let cardsHtml = '';

        for (let idx of indices) {
            const sig = signals[idx] || {};
            const sent = sentiment[idx] || {};
            const port = portfolios[idx] || {};

            const signalAction = sig.signal || 'NO_TRADE';
            const signalStrength = sig.signal_strength || signalAction;
            const signalClass = getSignalClass(signalStrength);
            const signalDisplay = formatSignalText(signalStrength);

            const sentimentScore = sig.sentiment_score !== undefined ? sig.sentiment_score : (sent.sentiment_score || 50);
            const sentimentLabel = sig.sentiment || sent.sentiment || 'Neutral';
            const spotPrice = sig.spot_price || 0;
            const cePrice = sig.ce_price || 0;
            const pePrice = sig.pe_price || 0;
            const pcr = sig.pcr || 1.0;
            const vix = sig.vix || 15.0;
            const confidence = sig.confidence || 0;

            // Multi-timeframe trends from market_sentiment (or signal object)
            const trend1m = sig.trend_1m || sent.trend_1m || 'NEUTRAL';
            const trend5m = sig.trend_5m || sent.trend_5m || 'NEUTRAL';
            const trend15m = sig.trend_15m || sent.trend_15m || 'NEUTRAL';

            // Score components
            const score1m = (sig.score_1m !== undefined) ? sig.score_1m : (sent.score_1m || 0);
            const score5m = (sig.score_5m !== undefined) ? sig.score_5m : (sent.score_5m || 0);
            const score15m = (sig.score_15m !== undefined) ? sig.score_15m : (sent.score_15m || 0);

            // Equity / active trade indicator
            const equity = port.equity || 100000;
            const openPos = port.open_positions || 0;

            cardsHtml += `
                <div class="card">
                    <div class="card-header">
                        <span class="index-name">${idx}</span>
                        <span class="signal-pill ${signalClass}">${signalDisplay}</span>
                    </div>
                    <div class="content">
                        <div class="metric-row">
                            <div class="metric"><div class="metric-label">SPOT</div><div class="metric-value">₹${formatNumber(spotPrice)}</div></div>
                            <div class="metric"><div class="metric-label">CE (₹)</div><div class="metric-value">${formatNumber(cePrice, 2)}</div></div>
                            <div class="metric"><div class="metric-label">PE (₹)</div><div class="metric-value">${formatNumber(pePrice, 2)}</div></div>
                        </div>
                        <div class="trends">
                            <div class="trend-badge ${getTrendClass(trend1m)}">📈 1m: ${trend1m} (${score1m})</div>
                            <div class="trend-badge ${getTrendClass(trend5m)}">⏱️ 5m: ${trend5m} (${score5m})</div>
                            <div class="trend-badge ${getTrendClass(trend15m)}">📊 15m: ${trend15m} (${score15m})</div>
                        </div>
                        <div class="sentiment-score" style="color: ${sentimentScore >= 70 ? '#a5ffb0' : (sentimentScore <= 30 ? '#ffaa99' : '#ffdd88')}">
                            ${formatNumber(sentimentScore, 1)}<span style="font-size:1rem;">/100</span>
                        </div>
                        <div class="sentiment-label">${sentimentLabel}</div>
                        <div class="metric-row">
                            <div class="metric"><div class="metric-label">PCR (OI)</div><div class="metric-value">${formatNumber(pcr, 2)}</div></div>
                            <div class="metric"><div class="metric-label">VIX</div><div class="metric-value">${formatNumber(vix, 1)}</div></div>
                            <div class="metric"><div class="metric-label">CONFIDENCE</div><div class="metric-value">${formatNumber(confidence, 0)}%</div></div>
                        </div>
                        <div class="footer-meta">
                            💰 Equity: ₹${formatNumber(equity, 0)} &nbsp;|&nbsp; 
                            ${openPos ? '🔓 ACTIVE POSITION' : '🔒 NO POSITION'}
                            ${sig.signal === 'ACTIVE' ? ' · trade running' : ''}
                        </div>
