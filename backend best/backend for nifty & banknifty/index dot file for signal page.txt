<script>
    // State Management
    let currentData = null;
    let equityHistory = [];
    let chart = null;
    let lastSignals = new Map();
    let audioContext = null;
    let soundEnabled = true;

    // Initialize Audio
    function initAudio() {
        if (!audioContext && soundEnabled) {
            audioContext = new (window.AudioContext || window.webkitAudioContext)();
        }
    }

    function playBeep(type) {
        if (!soundEnabled) return;
        try {
            initAudio();
            const oscillator = audioContext.createOscillator();
            const gainNode = audioContext.createGain();
            oscillator.connect(gainNode);
            gainNode.connect(audioContext.destination);
            oscillator.frequency.value = type === 'ce' ? 880 : 440;
            gainNode.gain.value = 0.3;
            oscillator.start();
            gainNode.gain.exponentialRampToValueAtTime(0.00001, audioContext.currentTime + 0.5);
            oscillator.stop(audioContext.currentTime + 0.5);
        } catch(e) { console.warn('Audio not supported'); }
    }

    function showToast(message, type) {
        const toast = document.getElementById('toast');
        const content = document.getElementById('toastContent');
        content.innerHTML = message;
        toast.className = `toast show ${type}`;
        setTimeout(() => {
            toast.classList.remove('show');
        }, 4000);
        if (type !== 'hold') playBeep(type);
    }

    // Format numbers
    function formatNumber(num, decimals = 2) {
        if (num === undefined || num === null) return '-';
        return num.toFixed(decimals);
    }

    function formatCurrency(num) {
        if (num === undefined || num === null) return '₹0';
        return '₹' + num.toLocaleString('en-IN', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    }

    function getSignalClass(signal) {
        if (signal === 'BUY_CE') return 'ce';
        if (signal === 'BUY_PE') return 'pe';
        return 'hold';
    }

    function getSignalText(signal) {
        if (signal === 'BUY_CE') return '🔵 BUY CE';
        if (signal === 'BUY_PE') return '🔴 BUY PE';
        return '⚪ HOLD';
    }

    // Update Portfolio UI
    function updatePortfolio(data) {
        const portfolio = data.portfolio || {};
        document.getElementById('equity').innerHTML = formatCurrency(portfolio.equity);
        const dailyPnl = portfolio.daily_pnl || 0;
        const pnlClass = dailyPnl >= 0 ? 'positive' : 'negative';
        document.getElementById('dailyPnl').innerHTML = `Today: <span class="${pnlClass}">${dailyPnl >= 0 ? '+' : ''}${formatCurrency(Math.abs(dailyPnl))}</span>`;
        document.getElementById('openPositions').innerHTML = portfolio.open_positions || 0;
        
        const currentPos = portfolio.current_position_index ? `${portfolio.current_position_index} ${portfolio.current_position_type}` : 'No active trade';
        document.getElementById('currentPosition').innerHTML = currentPos;
        
        let totalTrades = 0;
        let totalWins = 0;
        let bestIndex = '';
        let bestPnl = -Infinity;
        
        for (const [idx, sig] of Object.entries(data.index_signals || {})) {
            totalTrades += sig.daily_trades || 0;
            if (sig.daily_pnl > bestPnl) {
                bestPnl = sig.daily_pnl;
                bestIndex = idx;
            }
        }
        
        document.getElementById('totalTrades').innerHTML = totalTrades;
        const winRate = data.win_rate || 0;
        document.getElementById('winRate').innerHTML = winRate;
        
        if (bestIndex) {
            document.getElementById('bestPerformer').innerHTML = bestIndex;
            document.getElementById('bestPerformerPnl').innerHTML = `<span class="${bestPnl >= 0 ? 'positive' : 'negative'}">${bestPnl >= 0 ? '+' : ''}${formatCurrency(bestPnl)}</span>`;
        }
        
        // Update equity history for chart
        if (portfolio.equity) {
            equityHistory.push({ time: Date.now(), equity: portfolio.equity });
            if (equityHistory.length > 100) equityHistory.shift();
            updateChart();
        }
    }

    // Update Chart
    function updateChart() {
        if (!chart) return;
        const labels = equityHistory.map(h => new Date(h.time).toLocaleTimeString());
        const values = equityHistory.map(h => h.equity);
        chart.data.labels = labels;
        chart.data.datasets[0].data = values;
        chart.update();
    }

    // Check for new signals
    function checkNewSignals(data) {
        for (const [index, sig] of Object.entries(data.index_signals || {})) {
            const lastSignal = lastSignals.get(index);
            if (lastSignal !== sig.current_signal && sig.current_signal !== 'HOLD') {
                const message = `
                    <div style="display: flex; align-items: center; gap: 12px;">
                        <div style="font-size: 32px;">${sig.current_signal === 'BUY_CE' ? '📈' : '📉'}</div>
                        <div>
                            <strong style="font-size: 16px;">${index}</strong><br>
                            <span style="color: ${sig.current_signal === 'BUY_CE' ? '#00ff88' : '#ff4444'}">${sig.current_signal === 'BUY_CE' ? 'BUY CE' : 'BUY PE'}</span><br>
                            <small>Grade: ${sig.signal_grade} | Conf: ${sig.confidence}%</small>
                        </div>
                    </div>
                `;
                showToast(message, sig.current_signal === 'BUY_CE' ? 'ce' : 'pe');
            }
            lastSignals.set(index, sig.current_signal);
        }
    }

    // Render Index Cards
    function renderIndices(data) {
        const grid = document.getElementById('indicesGrid');
        const indices = ['NIFTY', 'BANKNIFTY', 'FINNIFTY', 'MIDCPNIFTY'];
        
        let html = '';
        for (const index of indices) {
            const sig = data.index_signals?.[index] || {};
            const signalClass = getSignalClass(sig.current_signal);
            const signalText = getSignalText(sig.current_signal);
            
            // RSI Color
            let rsiClass = '';
            if (sig.rsi > 70) rsiClass = 'rsi-high';
            else if (sig.rsi < 30) rsiClass = 'rsi-low';
            
            // MACD Color
            let macdClass = '';
            if (sig.macd_hist > 0) macdClass = 'macd-positive';
            else if (sig.macd_hist < 0) macdClass = 'macd-negative';
            
            // Active trade PnL
            let tradeHtml = '';
            let pnl = 0;
            if (sig.current_signal !== 'HOLD' && sig.entry_price > 0) {
                pnl = (sig.current_signal === 'BUY_CE' ? sig.ce_price - sig.entry_price : sig.pe_price - sig.entry_price);
                const pnlPercent = (pnl / sig.entry_price * 100);
                const pnlClass = pnl >= 0 ? 'positive' : 'negative';
                const progressPercent = Math.min(100, Math.max(0, ((pnl + (sig.target - sig.entry_price)) / (sig.target - sig.entry_price) * 100)));
                
                tradeHtml = `
                    <div class="trade-panel">
                        <div class="trade-header">
                            <span><i class="fas fa-sign-in-alt"></i> Active Position</span>
                            <span>Grade: ${sig.signal_grade || 'D'}</span>
                        </div>
                        <div>Entry: ${formatNumber(sig.entry_price)} | SL: ${formatNumber(sig.stop_loss)} | Target: ${formatNumber(sig.target)}</div>
                        <div class="trade-values">
                            <span>PnL: <span class="${pnlClass}">${pnl >= 0 ? '+' : ''}${formatNumber(pnl)} (${pnlPercent.toFixed(1)}%)</span></span>
                            <span>Peak: ${formatNumber(sig.max_profit_seen)}</span>
                        </div>
                        <div class="progress-bar">
                            <div class="progress-fill" style="width: ${progressPercent}%"></div>
                        </div>
                    </div>
                `;
            }
            
            html += `
                <div class="index-card">
                    <div class="card-header">
                        <div class="index-info">
                            <h3>${index}</h3>
                            <div class="expiry">${sig.expiry || 'N/A'} | Lot: ${sig.lot_size || '-'}</div>
                        </div>
                        <div class="spot-info">
                            <div class="spot-price">${formatNumber(sig.spot_price, 2)}</div>
                            <div class="spot-change">ATM: ${sig.atm_strike || '-'}</div>
                        </div>

    // Update Status Bar
    function updateStatus(data) {
        const wsRunning = data.ws_running;
        const wsLed = document.getElementById('wsLed');
        const wsStatus = document.getElementById('wsStatus');
        
        if (wsRunning) {
            wsStatus.innerHTML = 'Connected';
            wsLed.className = 'led green';
        } else {
            wsStatus.innerHTML = 'Disconnected';
            wsLed.className = 'led red';
        }
        
        document.getElementById('tickCount').innerHTML = data.tick_counter || 0;
        document.getElementById('lastUpdate').innerHTML = new Date().toLocaleTimeString();
    }

    // Fetch Data
    async function fetchData() {
        try {
            const response = await fetch('/api/all-signals');
            const data = await response.json();
            currentData = data;
            
            updatePortfolio(data);
            updateStatus(data);
            renderIndices(data);
            checkNewSignals(data);
            updateHistory(data);
        } catch (error) {
            console.error('Fetch error:', error);
            document.getElementById('wsStatus').innerHTML = 'Error';
            document.getElementById('wsLed').className = 'led red';
        }
    }
