def init_db():
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()
    
    # Ticks table with index field
    c.execute("""CREATE TABLE IF NOT EXISTS ticks (
        timestamp REAL, index_name TEXT, token TEXT, price REAL, 
        volume REAL, bid REAL, ask REAL, oi REAL)""")
    c.execute("CREATE INDEX IF NOT EXISTS idx_ticks_time_index ON ticks(timestamp, index_name)")
    
    # Signals table
    c.execute("""CREATE TABLE IF NOT EXISTS signals (
        timestamp REAL, index_name TEXT, action TEXT, signal_type TEXT, 
        grade TEXT, confidence REAL, entry_price REAL, stop_loss REAL, 
        target REAL, rsi REAL, pcr REAL, regime TEXT, factors TEXT)""")
    
    # Trades table
    c.execute("""CREATE TABLE IF NOT EXISTS trades (
        timestamp REAL, index_name TEXT, action TEXT, entry_price REAL, 
        exit_price REAL, pnl REAL, pnl_percent REAL, size_pct REAL, 
        status TEXT, grade TEXT, hold_time_min REAL)""")
    
    # Daily performance
    c.execute("""CREATE TABLE IF NOT EXISTS daily_performance (
        date TEXT, index_name TEXT, equity REAL, daily_pnl REAL, 
        trades_count INT, win_rate REAL, drawdown_pct REAL)""")
    
    conn.commit()
    conn.close()

init_db()

from SmartApi import SmartConnect

_original_parse = SmartWebSocketV2._parse_binary_data
def _patched_parse(self, binary_data):
    try:
        result = _original_parse(self, binary_data)
    except:
        result = {}
    try:
        token_bytes = binary_data[2:26]
        token_int = int.from_bytes(token_bytes, byteorder="little")
        result["token"] = str(token_int)
        ltp = int.from_bytes(binary_data[26:34], byteorder="little") / 100
        result["ltp"] = ltp
        volume = int.from_bytes(binary_data[34:42], byteorder="little")
        result["v"] = volume
    except Exception as e:
        pass
    return result

SmartWebSocketV2._parse_binary_data = _patched_parse

_original_on_close = SmartWebSocketV2._on_close
def _patched_on_close(self, wsapp, *args):
    try: _original_on_close(self, wsapp)
    except: pass
SmartWebSocketV2._on_close = _patched_on_close

def get_ist_now():
    return datetime.utcnow() + timedelta(hours=5, minutes=30)

def is_market_open():
    now_ist = get_ist_now()
    if now_ist.weekday() >= 5:
        return False
    market_start = dt_time(9, 15)
    market_end = dt_time(15, 30)
    return market_start <= now_ist.time() <= market_end

def get_auth_token():
    now = time.time()
    if auth_cache["token"] and (now - auth_cache["timestamp"] < 3600):
        return auth_cache["token"], auth_cache["feed_token"], auth_cache["obj"]
    try:
        totp = pyotp.TOTP(ANGEL_TOTP_SECRET).now()
        obj = SmartConnect(api_key=ANGEL_API_KEY)
        session = obj.generateSession(ANGEL_CLIENT_ID, ANGEL_PASSWORD, totp)
        if not session.get("status"):
            return None, None, None
        auth_token = session["data"]["jwtToken"]
        feed_token = obj.getfeedToken()
        auth_cache.update({"token": auth_token, "feed_token": feed_token, "timestamp": now, "obj": obj})
        return auth_token, feed_token, obj
    except Exception as e:
        logger.error(f"Auth failed: {e}")
        return None, None, None

def calculate_rsi(prices, period=14, smoothing=3):
    if len(prices) < period + 1:
        return 50.0
    gains, losses = [], []
    for i in range(1, len(prices)):
        diff = prices[i] - prices[i-1]
        gains.append(max(diff, 0))
        losses.append(max(-diff, 0))
    avg_gain = sum(gains[-period:]) / period
    avg_loss = sum(losses[-period:]) / period
    if avg_loss == 0:
        return 100.0
    rsi_raw = 100 - (100 / (1 + avg_gain / avg_loss))
    return rsi_raw

def calculate_macd(prices, fast=12, slow=26, signal=9):
    if len(prices) < slow + signal:
        return 0.0, 0.0, 0.0
    def ema(arr, p):
        alpha = 2 / (p + 1)
        val = arr[0]
        for x in arr[1:]:
            val = alpha * x + (1 - alpha) * val
        return val
    ema_fast = ema(prices[-fast:], fast)
    ema_slow = ema(prices[-slow:], slow)
    macd_line = ema_fast - ema_slow
    macd_history = []
    for i in range(signal, 0, -1):
        if len(prices) >= slow + i:
            ef = ema(prices[-(fast+i):-i], fast)
            es = ema(prices[-(slow+i):-i], slow)
            macd_history.append(ef - es)
    if macd_history:
        signal_line = ema(macd_history, signal)
    else:
        signal_line = macd_line
    histogram = macd_line - signal_line
    return macd_line, signal_line, histogram

def calculate_atr(prices, period=14):
    if len(prices) < period + 1:
        return 5.0
    tr_list = [abs(prices[i] - prices[i-1]) for i in range(1, len(prices))]
    return sum(tr_list[-period:]) / period

def calculate_adx(prices, period=14):
    if len(prices) < period * 2 + 1:
        return 20.0
    plus_dm, minus_dm, tr_list = [], [], []
    for i in range(1, len(prices)):
        up_move = prices[i] - prices[i-1]
        down_move = prices[i-1] - prices[i]
        plus_dm.append(max(up_move, 0) if up_move > down_move else 0)
        minus_dm.append(max(down_move, 0) if down_move > up_move else 0)
        tr_list.append(abs(prices[i] - prices[i-1]))
    atr = sum(tr_list[-period:]) / period
    plus_di = 100 * sum(plus_dm[-period:]) / period / atr if atr > 0 else 0
    minus_di = 100 * sum(minus_dm[-period:]) / period / atr if atr > 0 else 0
    dx = 100 * abs(plus_di - minus_di) / (plus_di + minus_di) if (plus_di + minus_di) > 0 else 0
    return dx

def calculate_ema(prices, period):
    if len(prices) < period:
        return prices[-1] if prices else 0
    alpha = 2 / (period + 1)
    ema = sum(prices[:period]) / period
    for p in prices[period:]:
        ema = alpha * p + (1 - alpha) * ema
    return ema

def get_pcr_from_oi(ce_oi_list, pe_oi_list):
    if ce_oi_list and pe_oi_list:
        ce_sum = sum(ce_oi_list)
        pe_sum = sum(pe_oi_list)
        if ce_sum > 0:
            return round(pe_sum / ce_sum, 2)
    return 1.0

def get_nifty_spot_ltp(token):
    _, _, obj = get_auth_token()
    if not obj:
        return None
    try:
        response = obj.ltpData("NSE", "NIFTY", token)
        if response.get("status") and response.get("data"):

    except:

def get_atm_tokens_for_index(index_config: IndexConfig):
    """Fetch ATM option tokens for a given index"""
    spot = get_nifty_spot_ltp(index_config.spot_token)
    if not spot:
        return None, None
    
    atm_strike = round(spot / index_config.min_strike_distance) * index_config.min_strike_distance
    
    try:
        url = "https://margincalculator.angelone.in/OpenAPI_File/files/OpenAPIScripMaster.json"
        resp = requests.get(url, timeout=15)
        df = pd.DataFrame(resp.json())
        
        # Filter for the specific index options
        opts = df[(df["name"] == index_config.symbol) & 
                  (df["instrumenttype"] == "OPTIDX") & 
                  (df["exch_seg"] == "NFO")].copy()
        
        if opts.empty:
            return None, None
        
        opts["expiry_date"] = pd.to_datetime(opts["expiry"], format="%d%b%Y", errors="coerce")
        opts = opts.dropna(subset=["expiry_date"])
        opts["strike"] = pd.to_numeric(opts["strike"], errors="coerce") / 100
        
        today = datetime.now()
        future = opts[opts["expiry_date"] >= today]
        if future.empty:
            return None, None
            
        nearest_expiry = future["expiry_date"].min()
        atm_opts = future[(future["strike"] == atm_strike) & (future["expiry_date"] == nearest_expiry)]
        
        ce = atm_opts[atm_opts["symbol"].str.contains("CE")]
        pe = atm_opts[atm_opts["symbol"].str.contains("PE")]
        
        if ce.empty or pe.empty:
            return None, None
            
        ce_token = str(ce.iloc[0]["token"])
        pe_token = str(pe.iloc[0]["token"])
        
        index_config.atm_strike = atm_strike
        index_config.expiry_date = nearest_expiry.strftime("%d%b%Y").upper()
        index_config.ce_token = ce_token
        index_config.pe_token = pe_token
        index_config.ce_symbol = ce.iloc[0]["symbol"]
        index_config.pe_symbol = pe.iloc[0]["symbol"]
        
        logger.info(f"{index_config.name} -> CE: {ce_token} | PE: {pe_token} | Strike: {atm_strike}")
        return ce_token, pe_token
        
    except Exception as e:
        logger.error(f"Error getting tokens for {index_config.name}: {e}")
        return None, None

def classify_signal_strength(data: IndexMarketData, index_config: IndexConfig, direction: str) -> Tuple[str, int, List[str]]:
    """Classify signal strength based on multiple factors"""
    score = 0
    factors = []
    
    if direction == "BULLISH":
        # RSI scoring
        if data.rsi >= CONFIG["EXTREME_CE_RSI"]:
            score += 3
            factors.append("RSI_EXTREME")
        elif data.rsi >= CONFIG["STRONG_CE_RSI"]:
            score += 2
            factors.append("RSI_STRONG")
        elif data.rsi >= CONFIG["CONSIDER_CE_RSI"]:
            score += 1
            factors.append("RSI_CONSIDER")
        
        # MACD scoring
        if data.macd_hist > CONFIG["MACD_CONFIRM_THRESHOLD"]:
            score += 2
            factors.append("MACD_CONFIRM")
        elif data.macd_hist > 0:
            score += 1
            factors.append("MACD_WEAK")
        
        # ADX trend strength
        if data.adx >= CONFIG["STRONG_TREND_THRESHOLD"]:
            score += 2
            factors.append("TREND_STRONG")
        elif data.adx >= CONFIG["MODERATE_TREND_THRESHOLD"]:
            score += 1
            factors.append("TREND_MODERATE")
        
        # PCR bullish
        if data.pcr <= CONFIG["PCR_BULLISH_THRESHOLD"]:
            score += 1
            factors.append("PCR_BULLISH")
        
        # Volume confirmation
        ce_vol_list = list(data.ce_volumes)
        if len(ce_vol_list) >= CONFIG["VOLUME_MA_PERIOD"]:
            avg_vol = sum(ce_vol_list[-CONFIG["VOLUME_MA_PERIOD"]:]) / CONFIG["VOLUME_MA_PERIOD"]
            current_vol = ce_vol_list[-1] if ce_vol_list else 0
            if current_vol > avg_vol * CONFIG["VOLUME_SPIKE_RATIO"]:
                score += 1
                factors.append("VOLUME_SPIKE")
        
        # EMA trend
        if data.spot_price > data.ema_fast:
            score += 1
            factors.append("PRICE_ABOVE_EMA20")
        if data.spot_price > data.ema_slow:
            score += 1
            factors.append("PRICE_ABOVE_EMA50")
            
    else:  # BEARISH
        if data.rsi <= CONFIG["EXTREME_PE_RSI"]:
            score += 3
            factors.append("RSI_EXTREME")
        elif data.rsi <= CONFIG["STRONG_PE_RSI"]:
            score += 2
            factors.append("RSI_STRONG")
        elif data.rsi <= CONFIG["CONSIDER_PE_RSI"]:
            score += 1
            factors.append("RSI_CONSIDER")
        
        if data.macd_hist < -CONFIG["MACD_CONFIRM_THRESHOLD"]:
            score += 2
            factors.append("MACD_CONFIRM")
        elif data.macd_hist < 0:
            score += 1
            factors.append("MACD_WEAK")
        
        if data.adx >= CONFIG["STRONG_TREND_THRESHOLD"]:
            score += 2
            factors.append("TREND_STRONG")
        elif data.adx >= CONFIG["MODERATE_TREND_THRESHOLD"]:
            score += 1
            factors.append("TREND_MODERATE")
        
        if data.pcr >= CONFIG["PCR_BEARISH_THRESHOLD"]:
            score += 1
            factors.append("PCR_BEARISH")
        
        pe_vol_list = list(data.pe_volumes)
        if len(pe_vol_list) >= CONFIG["VOLUME_MA_PERIOD"]:
            avg_vol = sum(pe_vol_list[-CONFIG["VOLUME_MA_PERIOD"]:]) / CONFIG["VOLUME_MA_PERIOD"]
            current_vol = pe_vol_list[-1] if pe_vol_list else 0
            if current_vol > avg_vol * CONFIG["VOLUME_SPIKE_RATIO"]:
                score += 1
                factors.append("VOLUME_SPIKE")
        
        if data.spot_price < data.ema_fast:
            score += 1
            factors.append("PRICE_BELOW_EMA20")
        if data.spot_price < data.ema_slow:
            score += 1
            factors.append("PRICE_BELOW_EMA50")
    
    # Determine strength
    if score >= 7:
        return "STRONG", score, factors
    elif score >= 5:
        return "CONSIDER", score, factors
    elif score >= 3:
        return "WEAK", score, factors
    else:
        return "VERY_WEAK", score, factors

def get_signal_grade(score: int) -> SignalGrade:
    if score >= 8:
        return SignalGrade.A_PLUS
    elif score >= 6:
        return SignalGrade.A
    elif score >= 5:
        return SignalGrade.B_PLUS
    elif score >= 4:
        return SignalGrade.B
    elif score >= 3:
        return SignalGrade.C
    else:
        return SignalGrade.D

def calculate_position_size(atr: float, equity: float, risk_percent: float) -> float:
    """Calculate position size based on ATR and risk"""
    if atr <= 0:
        return 0.05  # Default 5%
    risk_amount = equity * (risk_percent / 100)
    # Risk per lot = atr * lot_size (simplified)
    # Return position size as percentage of equity
    base_size = min(0.15, (risk_amount / (atr * 100)) * 0.01)
    return max(0.02, min(0.15, base_size))

def run_signal_engine_for_index(index_name: str):
    """Run signal generation for a specific index"""
    data = index_data[index_name]
    config = INDICES[index_name]
    
    if not config.enabled:
        return
    
    # Check if we have enough data
    spot_list = list(data.spot_prices)
    if len(spot_list) < CONFIG["MIN_DATA_POINTS"]:
        return
    
    # Skip if position exists for another index (can only hold one position at a time)
    if portfolio.open_positions > 0 and portfolio.current_position_index != index_name:
        return
    
    # Calculate indicators
    spot_prices = spot_list
    data.rsi = calculate_rsi(spot_prices, CONFIG["RSI_PERIOD"], CONFIG["RSI_SMOOTHING"])
    data.macd, data.macd_signal, data.macd_hist = calculate_macd(
        spot_prices, CONFIG["MACD_FAST"], CONFIG["MACD_SLOW"], CONFIG["MACD_SIGNAL"])
    data.atr = calculate_atr(spot_prices, CONFIG["ATR_PERIOD"])
    data.adx = calculate_adx(spot_prices, CONFIG["ADX_PERIOD"])
    data.ema_fast = calculate_ema(spot_prices, CONFIG["EMA_FAST"])
    data.ema_slow = calculate_ema(spot_prices, CONFIG["EMA_SLOW"])
    data.pcr = get_pcr_from_oi(list(data.ce_oi), list(data.pe_oi))
    data.spot_price = spot_prices[-1]
    
    current_time = time.time()
    price_above_fast = data.spot_price > data.ema_fast
    price_above_slow = data.spot_price > data.ema_slow
    
    # Check market hours for first/last minute avoidance
    now_ist = get_ist_now()
    minutes_from_start = (now_ist.hour - 9) * 60 + (now_ist.minute - 15)
    minutes_to_end = (15 * 60 + 30) - ((now_ist.hour - 9) * 60 + now_ist.minute)
    
    if minutes_from_start < CONFIG["AVOID_FIRST_LAST_MINUTES"] or minutes_to_end < CONFIG["AVOID_FIRST_LAST_MINUTES"]:
        return
    
    # Check volatility filter
    if data.atr < CONFIG["MIN_VOLATILITY_ATR"] or data.atr > CONFIG["MAX_VOLATILITY_ATR"]:
        return
    
    # EXISTING POSITION MANAGEMENT
    if data.current_signal in [SignalAction.BUY_CE, SignalAction.BUY_PE]:
        current_premium = data.ce_price if data.current_signal == SignalAction.BUY_CE else data.pe_price
        
        if current_premium == 0:
            return
        
        unrealized_pnl = current_premium - data.entry_price
        if unrealized_pnl > data.max_profit_seen:
            data.max_profit_seen = unrealized_pnl
        
        # Update trailing stop loss
        if current_premium > data.highest_premium:
            data.highest_premium = current_premium
            new_sl = current_premium - (data.atr * CONFIG["TRAILING_ATR_MULT"])
            if new_sl > data.stop_loss:
                data.stop_loss = new_sl
                if new_sl > data.entry_price and data.stop_loss <= data.entry_price:
                    logger.info(f"{index_name}: SL moved to breakeven")
        
        # Drawdown protection
        if data.max_profit_seen > 0:
            drawdown_from_peak = data.max_profit_seen - unrealized_pnl
            drawdown_pct = drawdown_from_peak / data.max_profit_seen if data.max_profit_seen > 0 else 0
            
            if drawdown_pct >= CONFIG["DRAWDOWN_ALERT_PCT"]:
                pnl_points = current_premium - data.entry_price
                pnl_value = pnl_points * config.lot_size
                portfolio.equity += pnl_value
                portfolio.daily_pnl += pnl_value
                data.daily_pnl += pnl_value
                data.daily_trades += 1
                portfolio.open_positions = 0
                data.current_signal = SignalAction.HOLD
                data.cooldown_until = current_time + CONFIG["COOLDOWN_SEC"]
                data.signal_buffer_ce = 0
                data.signal_buffer_pe = 0
                logger.info(f"{index_name}: EXIT - Drawdown protection. PnL: {pnl_value:.2f}")
                return
        
        # Stop loss hit
        if current_premium <= data.stop_loss:
            pnl_points = current_premium - data.entry_price
            pnl_value = pnl_points * config.lot_size
            portfolio.equity += pnl_value
            portfolio.daily_pnl += pnl_value
            data.daily_pnl += pnl_value
            data.daily_trades += 1
            portfolio.open_positions = 0
            data.current_signal = SignalAction.HOLD
            data.cooldown_until = current_time + CONFIG["COOLDOWN_SEC"]
            data.signal_buffer_ce = 0
            data.signal_buffer_pe = 0
            logger.info(f"{index_name}: EXIT - Stop loss. PnL: {pnl_value:.2f}")
            return
        
        # Target achieved
        if current_premium >= data.target:
            pnl_points = current_premium - data.entry_price
            pnl_value = pnl_points * config.lot_size
            portfolio.equity += pnl_value
            portfolio.daily_pnl += pnl_value
            data.daily_pnl += pnl_value
            data.daily_trades += 1
            portfolio.open_positions = 0
            data.current_signal = SignalAction.HOLD
            data.cooldown_until = current_time + CONFIG["COOLDOWN_SEC"]
            logger.info(f"{index_name}: EXIT - Target achieved. PnL: {pnl_value:.2f}")
            return
        
        # Max hold time
        hold_time_min = (current_time - data.entry_time) / 60
        if hold_time_min >= CONFIG["MAX_HOLD_TIME_MIN"]:
            pnl_points = current_premium - data.entry_price
            pnl_value = pnl_points * config.lot_size
            portfolio.equity += pnl_value
            portfolio.daily_pnl += pnl_value
            data.daily_pnl += pnl_value
            data.daily_trades += 1
            portfolio.open_positions = 0
            data.current_signal = SignalAction.HOLD
            data.cooldown_until = current_time + CONFIG["COOLDOWN_SEC"]
            logger.info(f"{index_name}: EXIT - Max hold time. PnL: {pnl_value:.2f}")
            return
        
        # Early exit on reversal
        if CONFIG["EARLY_EXIT_ON_REVERSAL"]:
            if data.current_signal == SignalAction.BUY_CE:
                if data.rsi < CONFIG["RSI_EXIT_CE"] or data.macd_hist < -CONFIG["MACD_EXIT_THRESHOLD"]:
                    pnl_points = current_premium - data.entry_price
                    pnl_value = pnl_points * config.lot_size
                    portfolio.equity += pnl_value
                    portfolio.daily_pnl += pnl_value
                    data.daily_pnl += pnl_value
                    data.daily_trades += 1
                    portfolio.open_positions = 0
                    data.current_signal = SignalAction.HOLD
                    data.cooldown_until = current_time + CONFIG["COOLDOWN_SEC"]
                    logger.info(f"{index_name}: EXIT - Momentum reversal (CE)")
                    return
            else:
                if data.rsi > CONFIG["RSI_EXIT_PE"] or data.macd_hist > CONFIG["MACD_EXIT_THRESHOLD"]:
                    pnl_points = current_premium - data.entry_price
                    pnl_value = pnl_points * config.lot_size
                    portfolio.equity += pnl_value
                    portfolio.daily_pnl += pnl_value
                    data.daily_pnl += pnl_value
                    data.daily_trades += 1
                    portfolio.open_positions = 0
                    data.current_signal = SignalAction.HOLD
                    data.cooldown_until = current_time + CONFIG["COOLDOWN_SEC"]
                    logger.info(f"{index_name}: EXIT - Momentum reversal (PE)")
                    return
        
        # Update status message
        hold_mins = int(hold_time_min)
        pnl_pct = ((current_premium - data.entry_price) / data.entry_price * 100) if data.entry_price > 0 else 0
        logger.debug(f"{index_name}: Active {data.current_signal.value} | Hold: {hold_mins}m | PnL: {pnl_pct:.1f}%")
        return
    
    # NEW SIGNAL GENERATION
    if current_time < data.cooldown_until:
        return
    
    # Check daily trade limit for this index
    if data.daily_trades >= CONFIG["max_daily_trades"]:
        return
    
    # Check consecutive same direction limit
    if data.consecutive_ce >= CONFIG["CONSECUTIVE_SAME_DIR_MAX"] and data.consecutive_pe >= CONFIG["CONSECUTIVE_SAME_DIR_MAX"]:
        return
    
    # CE Signal (Bullish)
    ce_conditions = (
        data.rsi >= CONFIG["CONSIDER_CE_RSI"] and
        data.macd_hist > 0 and
        price_above_fast
    )
    
    if ce_conditions and data.ce_price > 0:
        data.signal_buffer_ce += 1
        data.signal_buffer_pe = 0
        
        if data.signal_buffer_ce >= CONFIG["SIGNAL_BUILDING_TICKS"]:
            strength, score, factors = classify_signal_strength(data, config, "BULLISH")
            
            if strength != "VERY_WEAK" or CONFIG["WEAK_SIGNAL_ALLOWED"]:
                if data.consecutive_ce < CONFIG["CONSECUTIVE_SAME_DIR_MAX"]:
                    # Generate signal
                    sl = data.ce_price - (data.atr * CONFIG["ENTRY_ATR_MULT"])
                    target = data.ce_price + (data.atr * CONFIG["TARGET_ATR_MULT"])
                    position_size = calculate_position_size(data.atr, portfolio.equity, portfolio.risk_per_trade_pct)
                    
                    data.current_signal = SignalAction.BUY_CE
                    data.entry_price = data.ce_price
                    data.entry_time = current_time
                    data.stop_loss = sl
                    data.target = target
                    data.highest_premium = data.ce_price
                    data.max_profit_seen = 0
                    data.signal_grade = get_signal_grade(score)
                    data.confidence = min(95, score * 12 + data.rsi * 0.3)
                    data.last_signal_time = current_time
                    data.last_signal_type = "CE"
                    
                    data.consecutive_ce += 1
                    data.consecutive_pe = 0
                    portfolio.open_positions = 1
                    portfolio.current_position_index = index_name
                    portfolio.current_position_type = "CE"
                    
                    logger.info(f"🔵 {index_name}: {strength} CE BUY | Score: {score} | Grade: {data.signal_grade.value} | Conf: {data.confidence:.1f}%")
                    data.signal_buffer_ce = 0
                    return
    
    # PE Signal (Bearish)
    pe_conditions = (
        data.rsi <= CONFIG["CONSIDER_PE_RSI"] and
        data.macd_hist < 0 and
        not price_above_fast
    )
    
    if pe_conditions and data.pe_price > 0:
        data.signal_buffer_pe += 1
        data.signal_buffer_ce = 0
        
        if data.signal_buffer_pe >= CONFIG["SIGNAL_BUILDING_TICKS"]:
            strength, score, factors = classify_signal_strength(data, config, "BEARISH")
            
            if strength != "VERY_WEAK" or CONFIG["WEAK_SIGNAL_ALLOWED"]:
                if data.consecutive_pe < CONFIG["CONSECUTIVE_SAME_DIR_MAX"]:
                    sl = data.pe_price - (data.atr * CONFIG["ENTRY_ATR_MULT"])
                    target = data.pe_price + (data.atr * CONFIG["TARGET_ATR_MULT"])
                    
                    data.current_signal = SignalAction.BUY_PE
                    data.entry_price = data.pe_price
                    data.entry_time = current_time
                    data.stop_loss = sl
                    data.target = target
                    data.highest_premium = data.pe_price
                    data.max_profit_seen = 0
                    data.signal_grade = get_signal_grade(score)
                    data.confidence = min(95, score * 12 + (100 - data.rsi) * 0.3)
                    data.last_signal_time = current_time
                    data.last_signal_type = "PE"
                    
                    data.consecutive_pe += 1
                    data.consecutive_ce = 0
                    portfolio.open_positions = 1
                    portfolio.current_position_index = index_name
                    portfolio.current_position_type = "PE"
                    
                    logger.info(f"🔴 {index_name}: {strength} PE BUY | Score: {score} | Grade: {data.signal_grade.value} | Conf: {data.confidence:.1f}%")
                    data.signal_buffer_pe = 0
                    return
    
    # Reset buffers if no conditions met
    data.signal_buffer_ce = max(0, data.signal_buffer_ce - 1)
    data.signal_buffer_pe = max(0, data.signal_buffer_pe - 1)

def on_ws_data(wsapp, message):
    global tick_counter, last_tick_time
    
    last_tick_time = time.time()
    
    try:
        if isinstance(message, bytes):
            if sws is None:
                return
            tick = sws._parse_binary_data(message)
            if not tick:
                return
            ticks = [tick]
        else:
            data = json.loads(message) if isinstance(message, str) else message
            ticks = data if isinstance(data, list) else [data]
        
        for tick in ticks:
            token = str(tick.get("token") or tick.get("tk", ""))
            ltp = tick.get("ltp") or tick.get("last_traded_price", 0)
            
            if isinstance(ltp, (int, float)) and ltp > 50000:
                ltp = ltp / 100
            
            vol = tick.get("v") or tick.get("volume_trade_for_the_day", 0)
            oi = tick.get("oi") or tick.get("open_interest", 0)
            
            # Find which index this token belongs to
            for index_name, config in INDICES.items():
                data = index_data[index_name]
                
                if token == config.spot_token:
                    data.spot_price = ltp
                    data.spot_prices.append(ltp)
                    tick_counter += 1
                    
                    # Update timeframe candles
                    current_time = time.time()
                    for tf, interval in CONFIG["TIMEFRAME_SECONDS"].items():
                        candle = data.timeframe_candles[tf]
                        
                        if current_time - data.last_timeframe_update[tf] >= interval:
                            if candle["active"]:
                                data.timeframe_history[tf].append({
                                    "open": candle["open"],
                                    "high": candle["high"],
                                    "low": candle["low"],
                                    "close": candle["close"],
                                    "timestamp": data.last_timeframe_update[tf]
                                })
                            candle["open"] = ltp
                            candle["high"] = ltp
                            candle["low"] = ltp
                            candle["close"] = ltp
                            candle["active"] = True
                            data.last_timeframe_update[tf] = current_time
                        else:
                            if not candle["active"]:
                                candle["open"] = ltp
                                candle["low"] = ltp
                                candle["active"] = True
                            candle["high"] = max(candle["high"], ltp)
                            candle["low"] = min(candle["low"], ltp)
                            candle["close"] = ltp
                    
                elif token == config.ce_token:
                    data.ce_price = ltp
                    data.ce_prices.append(ltp)
                    data.ce_volumes.append(vol)
                    data.ce_oi.append(oi)
                    tick_counter += 1
                    
                elif token == config.pe_token:
                    data.pe_price = ltp
                    data.pe_prices.append(ltp)
                    data.pe_volumes.append(vol)
                    data.pe_oi.append(oi)
                    tick_counter += 1
            
            # Run signal engine for all indices periodically
            if tick_counter % (len(INDICES) * 2) == 0:
                for index_name in INDICES:
                    run_signal_engine_for_index(index_name)
                    
    except Exception as e:
        logger.error(f"WebSocket data error: {e}")

def on_ws_open(wsapp, open_message):
    logger.info(f"WebSocket opened: {open_message}")
    if sws:
        try:
            subscription_payload = []
            for config in INDICES.values():
                if config.ce_token and config.pe_token:
                    subscription_payload.append({"exchangeType": 1, "tokens": [config.spot_token]})
                    subscription_payload.append({"exchangeType": 2, "tokens": [config.ce_token, config.pe_token]})
            
            if subscription_payload:
                sws.subscribe("multi_index_trader", 1, subscription_payload)
                logger.info(f"Subscribed to {len(subscription_payload)} token groups")
        except Exception as e:
            logger.error(f"Subscription failed: {e}")

def on_ws_error(wsapp, error):
    logger.error(f"WebSocket Error: {error}")

def on_ws_close(wsapp, code, msg):
    global ws_running
    ws_running = False
    logger.warning(f"WebSocket closed: code={code}, msg={msg}")

def refresh_all_tokens():
    """Refresh option tokens for all indices"""
    for index_name, config in INDICES.items():
        if config.enabled:
            ce_token, pe_token = get_atm_tokens_for_index(config)
            if ce_token and pe_token:
                logger.info(f"Refreshed tokens for {index_name}")
            time.sleep(1)  # Rate limiting

def start_websocket():
    global CE_TOKEN, PE_TOKEN, sws, ws_running
    
    while True:
        try:
            if not is_market_open():
                time.sleep(30)
                continue
            
            # Refresh tokens for all indices
            refresh_all_tokens()
            
            auth_token, feed_token, obj = get_auth_token()
            if not feed_token:
                time.sleep(10)
                continue
            
            sws = SmartWebSocketV2(auth_token, ANGEL_API_KEY, ANGEL_CLIENT_ID, feed_token)
            
            sws.on_open = on_ws_open
            sws.on_data = on_ws_data
            sws.on_error = on_ws_error
            sws.on_close = on_ws_close
            
            ws_running = True
            sws.connect()
            
        except Exception as e:
            logger.error(f"WebSocket error: {e}")
            ws_running = False
            sws = None
            time.sleep(10)

@app.route("/api/all-signals")
def all_signals():
    result = {
        "timestamp": datetime.now().isoformat(),
        "ws_running": ws_running,
        "tick_counter": tick_counter,
        "portfolio": {
            "equity": portfolio.equity,
            "initial_equity": portfolio.initial_equity,
            "daily_pnl": portfolio.daily_pnl,
            "open_positions": portfolio.open_positions,
            "max_drawdown_today": portfolio.max_drawdown_today,
            "daily_loss_limit_pct": portfolio.daily_loss_limit_pct,
            "risk_per_trade_pct": portfolio.risk_per_trade_pct
        },
        "index_signals": {}
    }
    
    total_trades = 0
    winning_trades = 0
    
    for index_name, config in INDICES.items():
        data = index_data[index_name]
        
        # Calculate win rate approximation
        if data.daily_trades > 0:
            total_trades += data.daily_trades
            # Approximate winning trades based on daily PnL (simplified)
            if data.daily_pnl > 0:
                winning_trades += data.daily_trades
        
        result["index_signals"][index_name] = {
            "spot_price": data.spot_price,
            "ce_price": data.ce_price,
            "pe_price": data.pe_price,
            "rsi": round(data.rsi, 2),
            "macd_hist": round(data.macd_hist, 4),
            "adx": round(data.adx, 2),
            "atr": round(data.atr, 2),
            "pcr": round(data.pcr, 2),
            "ema_fast": round(data.ema_fast, 2),
            "current_signal": data.current_signal.value if data.current_signal else "HOLD",
            "signal_grade": data.signal_grade.value if data.signal_grade else "D",
            "confidence": round(data.confidence, 2),
            "entry_price": data.entry_price,
            "stop_loss": data.stop_loss,
            "target": data.target,
            "daily_trades": data.daily_trades,
            "daily_pnl": round(data.daily_pnl, 2),
            "entry_time": data.entry_time,
            "highest_premium": data.highest_premium,
            "max_profit_seen": round(data.max_profit_seen, 2),
            "cooldown_until": data.cooldown_until,
            "signal_buffer_ce": data.signal_buffer_ce,
            "signal_buffer_pe": data.signal_buffer_pe,
            "consecutive_ce": data.consecutive_ce,
            "consecutive_pe": data.consecutive_pe,
            "last_signal_time": data.last_signal_time,
            "last_signal_type": data.last_signal_type,
            "enabled": config.enabled,
            "lot_size": config.lot_size,
            "atm_strike": config.atm_strike,
            "expiry": config.expiry_date
        }
    
    result["total_trades"] = total_trades
    result["winning_trades"] = winning_trades
    result["win_rate"] = round((winning_trades / total_trades * 100), 1) if total_trades > 0 else 0
    
    return jsonify(result)

@app.route("/api/enable-index/<index_name>", methods=["POST"])
def enable_index(index_name):
    if index_name in INDICES:
        INDICES[index_name].enabled = True
        return jsonify({"status": "enabled", "index": index_name})
    return jsonify({"status": "error", "message": "Index not found"}), 404

@app.route("/api/disable-index/<index_name>", methods=["POST"])

    if index_name in INDICES:

        return jsonify({"status": "disabled", "index": index_name})
    return jsonify({"status": "error", "message": "Index not found"}), 404
